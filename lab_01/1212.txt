def updateElement():
    name = input("Please enter the name of the student to be updated: ")
    found = False
    for item in list:
        if item["name"] == name:
            found = True
            print("Current details of the student:")
            print(f"Name: {item['name']}, Phone: {item['phone']}, Age: {item['age']}, Address: {item['address']}")
            
            # Оновлення полів

            item["age"] = int(input("Enter new age (leave blank to keep current): ") or item["age"])
            item["address"] = input("Enter new address (leave blank to keep current): ") or item["address"]
            
            # Після оновлення сортуємо список
            list.sort(key=lambda x: x["name"])

    if not found:
